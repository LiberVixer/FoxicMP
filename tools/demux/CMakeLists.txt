cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(FoxicMPDemuxWorker LANGUAGES CXX)
endif()
find_path(FOXICMP_AVFORMAT_INCLUDE_DIR libavformat/avformat.h)
if(FOXICMP_AVFORMAT_INCLUDE_DIR)
 add_executable(foxicmp-demux-worker ../progressive_demux_worker.cpp)
 set_target_properties(foxicmp-demux-worker PROPERTIES AUTOMOC OFF CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
 target_include_directories(foxicmp-demux-worker PRIVATE ../../src ${FOXICMP_AVFORMAT_INCLUDE_DIR})
 if(MINGW)
  target_link_options(foxicmp-demux-worker PRIVATE -static)
 elseif(UNIX)
  target_link_libraries(foxicmp-demux-worker PRIVATE ${CMAKE_DL_LIBS})
 endif()
 install(TARGETS foxicmp-demux-worker RUNTIME DESTINATION .)
elseif(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 message(FATAL_ERROR "Supply matching FFmpeg headers using FOXICMP_AVFORMAT_INCLUDE_DIR")
endif()
